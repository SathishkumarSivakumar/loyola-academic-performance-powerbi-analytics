TABLE 4: Fact_Scores (Central Fact Table - 80K Rows)
--------------------------------------------------------------------------------
let
    Source = Csv.Document(File.Contents("C:\Users\SATHISHKUMAR\.gemini\antigravity\scratch\loyola_tn_project\Project\Loyola_TN_CSV_Package\scores.csv"), [Delimiter=",", Columns=17, Encoding=1252, QuoteStyle=QuoteStyle.None]),
    PromoteHeaders = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    ChangeTypes = Table.TransformColumnTypes(PromoteHeaders,{
        {"RollNo", type text},
        {"Program", type text},
        {"DegreeType", type text},
        {"Year", Int64.Type},
        {"Semester", Int64.Type},
        {"SubjectCode", type text},
        {"SubjectName", type text},
        {"ExamType", type text},
        {"ExamID", type text},
        {"RawMarks", type number},
        {"GraceMarks", type number},
        {"FinalMarks", type number},
        {"MaxMarks", type number},
        {"GradeLetter", type text},
        {"AttemptNo", Int64.Type},
        {"ReExamDate", type date},
        {"ExamDate", type date}
    }),
    AddGradePoint = Table.AddColumn(ChangeTypes, "GradePoints", each 
        if [GradeLetter] = "O" then 10
        else if [GradeLetter] = "A+" then 9
        else if [GradeLetter] = "A" then 8
        else if [GradeLetter] = "B+" then 7
        else if [GradeLetter] = "B" then 6
        else if [GradeLetter] = "C" then 5
        else 0, Int64.Type)
in
    AddGradePoint
